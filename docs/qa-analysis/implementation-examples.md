## 2. Security Testing Examples

### 2.1 SQL Injection Test Case Generation

```python
def generate_sql_injection_tests(input_fields: List[Dict]) -> List[ComprehensiveTestCase]:
    """Generate SQL injection test cases for discovered input fields"""
    
    sql_payloads = [
        "' OR '1'='1",
        "'; DROP TABLE users; --",
        "' UNION SELECT * FROM admin_users --",
        "1' OR '1'='1' ORDER BY 1 --",
        "'; EXEC xp_cmdshell('dir'); --"
    ]
    
    test_cases = []
    
    for field in input_fields:
        for payload in sql_payloads:
            test_case = ComprehensiveTestCase(
                test_id=f"SEC_SQL_{field['name']}_{hash(payload)}",
                scenario_name=f"SQL Injection Test - {field['name']} Field",
                metadata=TestCaseMetadata(
                    test_type="security",
                    priority="critical",
                    requirements_traceability=["SEC_REQ_001"],
                    risk_coverage=["sql_injection", "data_breach"]
                ),
                steps=[
                    EnhancedTestStep(
                        step_number=1,
                        action_type="type",
                        description=f"Enter SQL injection payload in {field['name']} field",
                        selector=field['selector'],
                        input_data=payload,
                        expected_result="Application properly sanitizes input and prevents SQL injection",
                        validation_rules=[
                            "No database error messages displayed",
                            "No unauthorized data access",
                            "Input is properly escaped or parameterized",
                            "Application logs security event"
                        ],
                        error_scenarios=[
                            "Database connection fails during injection attempt",
                            "WAF blocks malicious request"
                        ]
                    )
                ]
            )
            test_cases.append(test_case)
    
    return test_cases
```

## 5. Cross-Browser Testing Framework

### 5.1 Browser Compatibility Test Generation

```python
def generate_cross_browser_tests(base_test_case: ComprehensiveTestCase, browsers: List[BrowserTestConfig]) -> List[ComprehensiveTestCase]:
    """Generate browser-specific variations of test cases"""
    
    browser_specific_tests = []
    
    for browser_config in browsers:
        # Clone base test case
        browser_test = copy.deepcopy(base_test_case)
        
        # Modify for specific browser
        browser_test.test_id = f"{base_test_case.test_id}_{browser_config.browser}_{browser_config.version}"
        browser_test.scenario_name = f"{base_test_case.scenario_name} - {browser_config.browser.title()} {browser_config.version}"
        
        # Add browser-specific environment requirements
        browser_test.environment_requirements.update({
            "browser": browser_config.browser,
            "browser_version": browser_config.version,
            "operating_system": browser_config.operating_system,
            "viewport_size": f"{browser_config.viewport_size[0]}x{browser_config.viewport_size[1]}",
            "device_type": browser_config.device_type
        })
        
        # Add browser-specific validation rules
        for step in browser_test.steps:
            step.validation_rules.extend([
                f"Functionality works correctly in {browser_config.browser}",
                f"Visual rendering is consistent with design",
                f"No browser-specific console errors"
            ])
            
            # Add browser-specific performance expectations for mobile
            if browser_config.device_type == "mobile":
                step.performance_expectations = {
                    **step.performance_expectations,
                    "touch_response_time_ms": 100,
                    "scroll_performance_fps": 60
                }
        
        browser_specific_tests.append(browser_test)
    
    return browser_specific_tests
```

### 6.1 Boundary Value Test Data Generation

```python
class BoundaryValueGenerator:
    """Generate boundary value test data for various field types"""
    
    def generate_string_boundaries(self, min_length: int, max_length: int) -> Dict[str, List[str]]:
        """Generate boundary values for string fields"""
        return {
            "valid_boundaries": [
                "a" * min_length,  # Minimum valid length
                "a" * max_length,  # Maximum valid length
                "a" * (min_length + 1),  # Just above minimum
                "a" * (max_length - 1)   # Just below maximum
            ],
            "invalid_boundaries": [
                "",  # Empty string (if not allowed)
                "a" * (min_length - 1),  # Below minimum
                "a" * (max_length + 1),  # Above maximum
                "a" * (max_length * 2)   # Well above maximum
            ]
        }
    
    def generate_numeric_boundaries(self, min_val: int, max_val: int) -> Dict[str, List[int]]:
        """Generate boundary values for numeric fields"""
        return {
            "valid_boundaries": [
                min_val,           # Minimum valid value
                max_val,           # Maximum valid value
                min_val + 1,       # Just above minimum
                max_val - 1,       # Just below maximum
                (min_val + max_val) // 2  # Middle value
            ],
            "invalid_boundaries": [
                min_val - 1,       # Below minimum
                max_val + 1,       # Above maximum
                min_val - 100,     # Well below minimum
                max_val + 100      # Well above maximum
            ]
        }
    
    def generate_date_boundaries(self, min_date: str, max_date: str) -> Dict[str, List[str]]:
        """Generate boundary values for date fields"""
        min_dt = datetime.fromisoformat(min_date)
        max_dt = datetime.fromisoformat(max_date)
        
        return {
            "valid_boundaries": [
                min_date,
                max_date,
                (min_dt + timedelta(days=1)).isoformat(),
                (max_dt - timedelta(days=1)).isoformat()
            ],
            "invalid_boundaries": [
                (min_dt - timedelta(days=1)).isoformat(),
                (max_dt + timedelta(days=1)).isoformat(),
                "invalid-date-format",
                "2023-02-30"  # Invalid date
            ]
        }
```

                "user@.com",                  # Invalid domain

                "user name@example.com"       # Space in local part
